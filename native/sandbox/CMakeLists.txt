# yougame-sandbox: runs a game or emulator inside a Windows AppContainer + job object
# (launcher.cpp, see docs/DESKTOP_SANDBOX.md). sandbox-probe and steam-probe are test-only
# stand-ins for games; they are never installed or shipped.
# Built by desktop/scripts/build-libretro.mjs next to yougame-libretro. Windows only.
cmake_minimum_required(VERSION 3.24)
project(yougame_sandbox VERSION 1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
# The static CRT: the launcher needs nothing beside it.
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_executable(yougame-sandbox launcher.cpp)
target_compile_definitions(yougame-sandbox PRIVATE UNICODE _UNICODE NOMINMAX _WIN32_WINNT=0x0A00 WINVER=0x0A00)
target_compile_options(yougame-sandbox PRIVATE /utf-8 /W4 /WX)
# Resolve executable imports from System32 before wmain applies the same policy to later loads.
target_link_options(yougame-sandbox PRIVATE /DEPENDENTLOADFLAG:0x800)
target_link_libraries(yougame-sandbox PRIVATE userenv advapi32 user32 ole32 oleaut32 shell32 bcrypt fwpuclnt rpcrt4)
add_executable(sandbox-probe probe.cpp)
target_compile_definitions(sandbox-probe PRIVATE NOMINMAX _WIN32_WINNT=0x0A00 WINVER=0x0A00)
target_compile_options(sandbox-probe PRIVATE /utf-8 /W4 /WX)
target_link_libraries(sandbox-probe PRIVATE ws2_32 user32 gdi32 advapi32 shell32)
add_executable(steam-probe steam-probe.c)
set_property(TARGET steam-probe PROPERTY C_STANDARD 11)
target_compile_definitions(steam-probe PRIVATE UNICODE _UNICODE _WIN32_WINNT=0x0A00 WINVER=0x0A00)
target_compile_options(steam-probe PRIVATE /utf-8 /W4 /WX)
if(WIN32)
  enable_testing()
  add_executable(launcher-embed-test ../tests/launcher_embed.cpp)
  target_compile_definitions(launcher-embed-test PRIVATE _UNICODE _WIN32_WINNT=0x0A00 WINVER=0x0A00)
  if(MSVC)
    target_compile_options(launcher-embed-test PRIVATE /utf-8 /W4 /WX)
  endif()
  target_link_libraries(launcher-embed-test PRIVATE userenv advapi32 user32 ole32 oleaut32 shell32 bcrypt fwpuclnt rpcrt4)
  add_test(NAME launcher-embed COMMAND launcher-embed-test)
  set_tests_properties(launcher-embed PROPERTIES TIMEOUT 10)
  add_executable(launcher-children-test ../tests/launcher_children.cpp)
  target_compile_definitions(launcher-children-test PRIVATE UNICODE _UNICODE YG_DIRECT_BROKER=1 _WIN32_WINNT=0x0A00 WINVER=0x0A00)
  if(MSVC)
    target_compile_options(launcher-children-test PRIVATE /utf-8 /W4 /WX)
  endif()
  target_link_libraries(launcher-children-test PRIVATE userenv advapi32 user32 ole32 oleaut32 shell32 bcrypt fwpuclnt rpcrt4)
  add_test(NAME launcher-children-default COMMAND launcher-children-test 0)
  add_test(NAME launcher-children-one COMMAND launcher-children-test 1)
  add_test(NAME launcher-children-two COMMAND launcher-children-test 2)
  add_test(NAME launcher-children-options COMMAND launcher-children-test options)
  add_test(NAME launcher-children-write-execute COMMAND launcher-children-test write-execute)
  set_tests_properties(launcher-children-default launcher-children-one launcher-children-two launcher-children-options launcher-children-write-execute PROPERTIES TIMEOUT 30)
endif()
install(TARGETS yougame-sandbox RUNTIME DESTINATION .)

if("$ENV{YOUGAME_BUILD_DIRECT_HELPER}" STREQUAL "1")
  target_compile_definitions(yougame-sandbox PRIVATE YG_DIRECT_BROKER=1)
endif()
